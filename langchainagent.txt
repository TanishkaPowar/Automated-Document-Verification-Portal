"""
LangChain Tool Binding with Groq API
Demonstrates addition and subtraction using bind_tools
"""

from langchain_groq import ChatGroq
from langchain_core.tools import tool
from langchain_core.messages import HumanMessage, ToolMessage
import os

# ─────────────────────────────────────────────
# 1. Define Tools
# ─────────────────────────────────────────────

@tool
def add(a: float, b: float) -> float:
    """Add two numbers together and return the result."""
    return a + b


@tool
def subtract(a: float, b: float) -> float:
    """Subtract b from a and return the result."""
    return a - b


# ─────────────────────────────────────────────
# 2. Initialize Groq LLM & Bind Tools
# ─────────────────────────────────────────────

llm = ChatGroq(
    model="llama3-8b-8192",          # or "mixtral-8x7b-32768", "llama3-70b-8192"
    temperature=0,
    api_key=os.environ.get("GROQ_API_KEY"),  # set your key here or via env var
)

tools = [add, subtract]
llm_with_tools = llm.bind_tools(tools)   # ← bind_tools attaches tool schemas to the model

# ─────────────────────────────────────────────
# 3. Tool Execution Helper
# ─────────────────────────────────────────────

tool_map = {t.name: t for t in tools}

def run_tool_call(tool_call: dict) -> ToolMessage:
    """Execute a single tool call and return a ToolMessage."""
    name = tool_call["name"]
    args = tool_call["args"]
    result = tool_map[name].invoke(args)
    return ToolMessage(
        content=str(result),
        tool_call_id=tool_call["id"],
    )

# ─────────────────────────────────────────────
# 4. Full Agentic Loop
# ─────────────────────────────────────────────

def run_agent(user_query: str) -> str:
    """
    Send a query to the LLM, execute any tool calls it requests,
    then return the final natural-language answer.
    """
    print(f"\n{'='*50}")
    print(f"User: {user_query}")

    messages = [HumanMessage(content=user_query)]

    # Step 1: First LLM call — may return tool_calls
    response = llm_with_tools.invoke(messages)
    messages.append(response)

    # Step 2: Execute tool calls if the model requested any
    if response.tool_calls:
        print(f"\n[Tool calls requested: {[tc['name'] for tc in response.tool_calls]}]")
        for tool_call in response.tool_calls:
            tool_msg = run_tool_call(tool_call)
            print(f"  → {tool_call['name']}({tool_call['args']}) = {tool_msg.content}")
            messages.append(tool_msg)

        # Step 3: Second LLM call — generate final answer using tool results
        final_response = llm_with_tools.invoke(messages)
        answer = final_response.content
    else:
        answer = response.content

    print(f"\nAssistant: {answer}")
    return answer


# ─────────────────────────────────────────────
# 5. Run Example Queries
# ─────────────────────────────────────────────

if __name__ == "__main__":
    queries = [
        "What is 42 + 58?",
        "Subtract 15 from 100.",
        "If I have 200 and spend 75, how much is left?",
        "What is (30 + 20) - 10?",   # multi-step: LLM may chain two tool calls
    ]

    for query in queries:
        run_agent(query)
